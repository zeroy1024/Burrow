// This file is compiled only through the README gallery overlay.
func runREADMEGallery() error {
	root := filepath.Dir(os.Args[0])
	directory := filepath.Join(root, "..", "..", "..", "gallery-data")
	if err := os.MkdirAll(directory, 0700); err != nil {
		return err
	}
	mygo.App.SetName("Burrow Gallery")
	mygo.App.SetPath(mygo.PathUserData, directory)
	mygo.App.SetPath(mygo.PathCache, filepath.Join(directory, "cache"))
	mygo.App.SetPath(mygo.PathLogs, filepath.Join(directory, "logs"))
	mygo.App.SetLanguages("zh-Hans", "en")
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	db, err := store.Open(ctx, filepath.Join(directory, "burrow.db"))
	if err != nil {
		return err
	}
	defer db.Close()
	a := &app{ctx: ctx, cancel: cancel, store: db, dataDirectory: directory, statusVisible: true}
	a.persistence = newPersistenceWorker(func() *mygo.Window { return a.window })
	defer a.finishPersistence()
	if err := a.loadTerminalSettings(); err != nil {
		return err
	}
	if err := a.loadAppPreferences(); err != nil {
		return err
	}
	a.security = applicationSecurityState{ready: true, initialized: true}
	a.prefs.ReduceMotion = "on"
	a.prefs.Translucency = false
	a.terminal.prefs.CursorBlink = false
	a.terminal.prefs.FontSize = 13
	a.terminal.prefs.LineHeight = 21
	for i := range a.pages {
		a.pages[i].sidebarWidth = 236
	}
	a.hosts, a.groups = demoHosts(), demoGroups()
	for i := range a.hosts {
		a.hosts[i].address = fmt.Sprintf("192.0.2.%d", 10+i)
		a.hosts[i].user = "deploy"
		initializeSession(&a.hosts[i])
	}
	a.hosts[0].system = "Ubuntu 24.04 LTS"
	a.hosts[7].name, a.hosts[7].user = "home-lab", "ops"
	a.hosts[8].name = "edge-tokyo"
	for _, group := range a.groups {
		group.stableID = newSessionID()
	}
	for i := range 4 {
		h := a.hosts[i]
		h.protocol = "sftp"
		h.session.ID = newSessionID()
		a.groups[3].hosts = append(a.groups[3].hosts, hostID(len(a.hosts)))
		a.hosts = append(a.hosts, h)
	}
	a.galleryTerminals()
	a.galleryFiles()
	a.galleryMCP()
	stopTheme := mygo.Theme.OnUpdated(func() { a.syncAppearance() })
	defer stopTheme()
	mygo.Theme.SetSource(mygo.ThemeDark)
	mygo.App.WhenReady(func() {
		a.syncAppearance()
		a.window = mygo.NewWindow(mygo.WindowOptions{
			Title: "Burrow", Width: 1424, Height: 900,
			MinWidth: 1180, MinHeight: 760,
			TitleBarStyle: mygo.TitleBarHidden, TitleBarHeight: 44,
			TrafficLightPosition: &mygo.Point{X: 20, Y: 22},
			Content:              ui.View(a.view),
		})
		var scene = "ssh-dark"
		choose := func(name string, section section, source mygo.ThemeSource) func(*mygo.MenuItem, *mygo.Window) {
			return func(*mygo.MenuItem, *mygo.Window) {
				scene = name
				a.section = section
				a.settingsUI.open = false
				mygo.Theme.SetSource(source)
				a.syncAppearance()
				a.window.Invalidate()
			}
		}
		mygo.App.SetMenu(mygo.NewMenu([]*mygo.MenuItem{
			{Label: "Burrow Gallery", Submenu: []*mygo.MenuItem{{Role: mygo.RoleQuit}}},
			{Label: "展示场景", Submenu: []*mygo.MenuItem{
				{Label: "SSH · 深色", Accelerator: "CmdOrCtrl+1", Click: choose("ssh-dark", sectionSSH, mygo.ThemeDark)},
				{Label: "SSH · 浅色", Accelerator: "CmdOrCtrl+2", Click: choose("ssh-light", sectionSSH, mygo.ThemeLight)},
				{Label: "文件传输", Accelerator: "CmdOrCtrl+3", Click: choose("files", sectionFiles, mygo.ThemeLight)},
				{Label: "MCP 审计", Accelerator: "CmdOrCtrl+4", Click: choose("mcp", sectionMCP, mygo.ThemeLight)},
				mygo.Separator(),
				{Label: "保存当前展示图", Accelerator: "CmdOrCtrl+Shift+S", Click: func(*mygo.MenuItem, *mygo.Window) {
					data, err := a.window.CapturePage()
					if err == nil {
						output := filepath.Join(galleryScreenshotDirectory, scene+".png")
						err = os.WriteFile(output, data, 0644)
					}
					if err != nil {
						fmt.Fprintln(os.Stderr, "Gallery capture:", err)
					}
				}},
			}},
		}))
		a.window.Show()
	})
	return mygo.App.Run()
}

func galleryLines(text []string) []terminalLine {
	lines := []terminalLine{}
	for _, value := range text {
		span := terminalSpan{Text: value}
		if len(value) > 0 && value[0] == '$' {
			span.Foreground, span.Bold = terminalIndexed(4), true
		}
		lines = append(lines, terminalLine{span})
	}
	return lines
}
